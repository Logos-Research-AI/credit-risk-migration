520        proc logistic data=work.model_features_train descending;
521            model default_flag =
522                /* Core risk metrics */
523                credit_utilization_capped debt_ratio_capped
524

530                age monthly_income log_monthly_income

533        
534                /* Risk flags */
535                flag_high_utilization flag_very_high_utilization
536                flag_high_debt_ratio flag_very_high_debt_ratio
537                flag_recent_late_payments flag_serious_delinquency
538                flag_multiple_delinquencies flag_low_income
539                flag_many_dependents flag_few_credit_lines
540                total_risk_flags
541        
542                /* Categorical indicators */
543                credit_util_category debt_category age_category
544        
545                /* Composite scores */
546                financial_health
547        
548                /* Interaction features */
549                age_income_interaction utilization_debt_interaction
550        
551                /* Missing value flags */
552                MonthlyIncome_missing Dependents_missing
553        
554            / selection=stepwise slentry=0.05 slstay=0.05
555              ctable lackfit rsquare;
556        
557            /* Output predictions */
558            output out=work.logit_scored_train
559                   predicted=pd_logistic;
560

562            store work.logit_model;

594            if default_flag = 1 and pred_logistic = 1 then TP = count;
