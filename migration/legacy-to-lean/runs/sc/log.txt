325        data work.ks_calculation;
326            set work.logit_scored_valid;
327            retain cum_good cum_bad;
328        
329            if _n_ = 1 then do;
330                cum_good = 0;

355        /*****************************************************************************
356         * 4. Confusion Matrix and Accuracy Metrics
357         *****************************************************************************/
358        
359        /* Calculate metrics at different thresholds */
360        %macro confusion_metrics(threshold, suffix);
361            data work.confusion_&suffix;
362                set work.logit_scored_valid;
363                predicted = (pd_logistic >= &threshold);
364            run;
365        
366            proc freq data=work.confusion_&suffix noprint;
367                tables default_flag * predicted / out=work.conf_matrix_&suffix;
368            run;
369        
370            /* Calculate metrics */
371            data work.metrics_&suffix;
372                set work.conf_matrix_&suffix end=eof;
373                retain TP TN FP FN 0;
374        
375                if default_flag = 1 and predicted = 1 then TP = count;
376                else if default_flag = 0 and predicted = 0 then TN = count;
377                else if default_flag = 0 and predicted = 1 then FP = count;
378                else if default_flag = 1 and predicted = 0 then FN = count;
379        
380                if eof then do;
381                    threshold = &threshold;
382                    if (TP + TN + FP + FN) > 0 then accuracy = (TP + TN) / (TP + TN + FP + FN);
383                    else accuracy = .;
384                    if (TP + FP) > 0 then precision = TP / (TP + FP);
385                    else precision = .;
386                    if (TP + FN) > 0 then recall = TP / (TP + FN);
387                    else recall = .;
388                    if (precision + recall) > 0 then f1_score = 2 * (precision * recall) / (precision + recall);
389                    else f1_score = .;
390                    if (TN + FP) > 0 then specificity = TN / (TN + FP);
391                    else specificity = .;
392                    output;
393                end;
394                keep threshold accuracy precision recall f1_score specificity;
395            run;
396        %mend;

502        /* Calculate PSI */
503        proc sql;
504            create table work.psi_calculation as
505            select
506                a.score_bin,
507                a.train_pct,
508                b.valid_pct,
509                (a.train_pct - b.valid_pct) * log(a.train_pct / b.valid_pct) as psi_component
510            from
511                (select score_bin, count(*) / (select count(*) from work.train_bins) as train_pct
512                 from work.train_bins group by score_bin) a
513            join
514                (select score_bin, count(*) / (select count(*) from work.valid_bins) as valid_pct
515                 from work.valid_bins group by score_bin) b
516            on a.score_bin = b.score_bin;
517        
518            create table work.psi_summary as
519            select sum(psi_component) as PSI format=8.4,
520                   case
521                      when calculated PSI < 0.1 then 'No significant change'
522                      when calculated PSI < 0.25 then 'Some change'
523                      else 'Significant change'
524                   end as PSI_interpretation
525            from work.psi_calculation;
526        quit;

529            title "Population Stability Index";
