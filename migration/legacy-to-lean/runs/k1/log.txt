331        data work.ks_calculation;
332            set work.logit_scored_valid;
333            retain cum_good cum_bad;
334        
335            if _n_ = 1 then do;
336                cum_good = 0;

361        /*****************************************************************************
362         * 4. Confusion Matrix and Accuracy Metrics
363         *****************************************************************************/
364        
365        /* Calculate metrics at different thresholds */
366        %macro confusion_metrics(threshold, suffix);
367            data work.confusion_&suffix;
368                set work.logit_scored_valid;
369                predicted = (pd_logistic >= &threshold);
370            run;
371        
372            proc freq data=work.confusion_&suffix noprint;
373                tables default_flag * predicted / out=work.conf_matrix_&suffix;
374            run;
375        
376            /* Calculate metrics */
377            data work.metrics_&suffix;
378                set work.conf_matrix_&suffix end=eof;
379                retain TP TN FP FN 0;
380        
381                if default_flag = 1 and predicted = 1 then TP = count;
382                else if default_flag = 0 and predicted = 0 then TN = count;
383                else if default_flag = 0 and predicted = 1 then FP = count;
384                else if default_flag = 1 and predicted = 0 then FN = count;
385        
386                if eof then do;
387                    threshold = &threshold;
388                    if (TP + TN + FP + FN) > 0 then accuracy = (TP + TN) / (TP + TN + FP + FN);
389                    else accuracy = .;
390                    if (TP + FP) > 0 then precision = TP / (TP + FP);
391                    else precision = .;
392                    if (TP + FN) > 0 then recall = TP / (TP + FN);
393                    else recall = .;
394                    if (precision + recall) > 0 then f1_score = 2 * (precision * recall) / (precision + recall);
395                    else f1_score = .;
396                    if (TN + FP) > 0 then specificity = TN / (TN + FP);
397                    else specificity = .;
398                    output;
399                end;
400                keep threshold accuracy precision recall f1_score specificity;
401            run;
402        %mend;

508        /* Calculate PSI */
509        proc sql;
510            create table work.psi_calculation as
511            select
512                a.score_bin,
513                a.train_pct,
514                b.valid_pct,
515                (a.train_pct - b.valid_pct) * log(a.train_pct / b.valid_pct) as psi_component
516            from
517                (select score_bin, count(*) / (select count(*) from work.train_bins) as train_pct
518                 from work.train_bins group by score_bin) a
519            join
520                (select score_bin, count(*) / (select count(*) from work.valid_bins) as valid_pct
521                 from work.valid_bins group by score_bin) b
522            on a.score_bin = b.score_bin;
523        
524            create table work.psi_summary as
525            select sum(psi_component) as PSI format=8.4,
526                   case
527                      when calculated PSI < 0.1 then 'No significant change'
528                      when calculated PSI < 0.25 then 'Some change'
529                      else 'Significant change'
530                   end as PSI_interpretation
531            from work.psi_calculation;
532        quit;

535            title "Population Stability Index";
