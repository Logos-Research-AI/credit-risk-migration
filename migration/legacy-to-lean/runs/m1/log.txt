649            data work.confusion_&suffix;

658            /* Calculate metrics */
659            data work.metrics_&suffix;
660                set work.conf_matrix_&suffix end=eof;
661                retain TP TN FP FN 0;
662        
663                if default_flag = 1 and predicted = 1 then TP = count;
664                else if default_flag = 0 and predicted = 0 then TN = count;
665                else if default_flag = 0 and predicted = 1 then FP = count;
666                else if default_flag = 1 and predicted = 0 then FN = count;
667        
668                if eof then do;
669                    threshold = &threshold;
670                    if (TP + TN + FP + FN) > 0 then accuracy = (TP + TN) / (TP + TN + FP + FN);
671                    else accuracy = .;
672                    if (TP + FP) > 0 then precision = TP / (TP + FP);
673                    else precision = .;
674                    if (TP + FN) > 0 then recall = TP / (TP + FN);
675                    else recall = .;
676                    if (precision + recall) > 0 then f1_score = 2 * (precision * recall) / (precision + recall);
677                    else f1_score = .;
678                    if (TN + FP) > 0 then specificity = TN / (TN + FP);
679                    else specificity = .;
680                    output;
681                end;
682                keep threshold accuracy precision recall f1_score specificity;
683            run;

790        /* Calculate PSI */
791        proc sql;
792            create table work.psi_calculation as
793            select
794                a.score_bin,
795                a.train_pct,
796                b.valid_pct,
797                (a.train_pct - b.valid_pct) * log(a.train_pct / b.valid_pct) as psi_component
798            from
799                (select score_bin, count(*) / (select count(*) from work.train_bins) as train_pct
800                 from work.train_bins group by score_bin) a
801            join
802                (select score_bin, count(*) / (select count(*) from work.valid_bins) as valid_pct
803                 from work.valid_bins group by score_bin) b
804            on a.score_bin = b.score_bin;
805        
806            create table work.psi_summary as
807            select sum(psi_component) as PSI format=8.4,
808                   case
809                      when calculated PSI < 0.1 then 'No significant change'
810                      when calculated PSI < 0.25 then 'Some change'
811                      else 'Significant change'
812                   end as PSI_interpretation
813            from work.psi_calculation;
